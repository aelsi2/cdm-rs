1//! Startup code and minimal runtime for the CDM-16 processor.
2//!
3//! # Features
4//!
5//! This crate provides:
6//! - Default interrupt and exception handlers using the `halt` instruction
7//! - A macro for defining the entry point of the program: [`entry`](macro@entry)
8//! - A macro for redefining exception handlers: [`exception`](macro@exception)
9//! - A macro for defining interrupt handlers: [`interrupt`](macro@interrupt)
10//! - A macro for registering interrupt handlers in the IVT:
11//! [`interrupt_vectors`](macro@interrupt_vectors)
12//!
13//! # Cargo features
14//! #### `interrupts`
15//! This feature disables the default interrupt handlers and allows the use of [`interrupt_vectors`](macro@interrupt_vectors)
16//!
17//! #### `harvard`
18//! This feature enables generation of images for the [Harvard architecture](https://en.wikipedia.org/wiki/Harvard_architecture) with separate program and data address spaces.
19//! By default, the [von Neumann architecture](https://en.wikipedia.org/wiki/Von_Neumann_architecture) with one address space is used.
20//!
21//! # Requirements
22//! #### Rust flags
23//! The crate generates a linker script named `link.x` in the output directory. It needs to be passed to the
24//! linker. This can be done by passing `-Clink-arg=-Tlink.x` to rustc.
25//!
26//! #### Entry point
27//! Exactly one function needs to be marked as the application entry point by applying the [`entry`](macro@entry) attribute.
28//! ***
29//!
30//! #### Example
31//! ##### `.cargo/config.toml`
32//! ```toml
33//! [build]
34//! target = "cdm-none"
35//! rustflags = [ "-Clink-arg=-Tlink.x" ]
36//! [unstable]
37//! build-std = [ "core" ]
38//! ```
39//!
40//! ##### `src/main.rs`
41//! ```rust
42//! #![no_std]
43//! #![no_main]
44//!
45//! use cdm_rt::entry;
46//!
47//! #[entry]
48//! fn main() -> ! {
49//!     loop { /* .. */ }
50//! }
51//! ```
52//!
53//! ## Memory mapped I/O
54//! If you want to use memory mapped I/O, you will need to:
55//! - create a linker script (e.g. `memory.x` in the project root)
56//! - set the value of `RAM_ORIGIN` and/or `RAM_LENGTH` in the linker script to reserve a range of addresses for the MMIO registers
57//! - create one or more symbols for the registers and set their values in the linker script
58//! - pass the script to the linker (e.g. pass `-Clink-arg=-Tmemory.x` to rustc)
59//! - declare the symbols in Rust inside an `extern "C"` block
60//! - use [`core::ptr::read_volatile`](https://doc.rust-lang.org/core/ptr/fn.read_volatile.html) and/or
61//! [`core::ptr::write_volatile`](https://doc.rust-lang.org/core/ptr/fn.write_volatile.html) to read from and write
62//! to the registers
63//!
64//! **Important note:** when using the von Neumann architecture, `RAM_ORIGIN` and MMIO register
65//! addresses need to be greater or equal to `0x100` (256 in decimal) to avoid overlapping with the IVT.
66//! ***
67//!
68//! #### Example
69//! ##### `.cargo/config.toml`
70//! ```toml
71//! [build]
72//! target = "cdm-none"
73//! rustflags = [ "-Clink-arg=-Tlink.x", "-Clink-arg=-Tmemory.x" ]
74//! [unstable]
75//! build-std = [ "core" ]
76//! ```
77//!
78//! ##### `memory.x`
79//! ```ld
80//! RAM_ORIGIN = 0x120;
81//!
82//! MMIO_IN = 0x100;
83//! MMIO_OUT = 0x102;
84//! ```
85//!
86//! ##### `src/mmio.rs`
87//! ```rust
88//! unsafe extern "C" {
89//!     static MMIO_IN: u16;
90//!     static mut MMIO_OUT: u16;
91//! }
92//!
93//! pub fn get_in() -> u16 {
94//!     unsafe { core::ptr::read_volatile(&raw MMIO_IN) }
95//! }
96//!
97//! pub fn set_out(value: u16) {
98//!     unsafe { core::ptr::write_volatile(&raw mut MMIO_OUT, value) }
99//! }
100//! ```

115/// The number of exception vectors in the interrupt vector table, including the start vector.
116pub const EXCEPTION_COUNT: usize = 16;

118/// The number of application-specific interrupt vectors in the interrupt vector table.
119pub const INTERRUPT_COUNT: usize = 64 - EXCEPTION_COUNT;

121/// Represents a vector in the interrupt vector table.
122///
123/// The first field is the pointer to the handler function.
124/// The second field is the initial value of the processor status register.
125///
126/// Use `#[interrupt]` to define interrupt handler functions.
127#[derive(Clone, Copy)]
128#[repr(C)]
129pub struct InterruptVector(pub unsafe extern "cdm-isr" fn(), pub Psr);
130
131impl InterruptVector {
132    /// The default vector used in absence of an explicit definition.
133    ///
134    /// Calls `_interrupt_handler`, which triggers a hardware halt by default.
135    pub const DEFAULT: Self = {
136        unsafe extern "cdm-isr" {
137            fn _interrupt_handler();
138        }
139        InterruptVector(_interrupt_handler, Psr::NONE)
140    };
141}

143/// Defines the application-specific interrupt handler section of the interrupt vector table.
144///
145/// The interrupt vectors specified in the arguments are placed sequentially after exception vectors,
146/// starting from index `EXCEPTION_COUNT`. The rest of the table is filled with `InterruptVector::DEFAULT`.
147///
148/// Must be used **once** in the dependency graph.
149///
150/// Use the [`interrupt`](macro@interrupt) attribute to define interrupt handler functions.
151///
152/// ``` no_run
153/// interrupt_vectors![
154///     InterruptVector(MyHandler1, Psr::NONE), // int EXCEPTION_COUNT+0
155///     InterruptVector(MyHandler2, Psr::NONE), // int EXCEPTION_COUNT+1
156///     InterruptVector(MyHandler3, Psr::NONE), // int EXCEPTION_COUNT+2
157/// ];
158///
159/// #[cdm_rt::interrupt]
160/// fn MyHandler1() { /* .. */ }
161/// #[cdm_rt::interrupt]
162/// fn MyHandler2() { /* .. */ }
163/// #[cdm_rt::interrupt]
164/// fn MyHandler3() { /* .. */ }
165/// ```
166#[cfg(feature = "interrupts")]
167#[macro_export]
168macro_rules! interrupt_vectors {
169    ($($elems:expr),* $(,)?) => {
170        #[used]
171        #[unsafe(no_mangle)]
172        #[unsafe(link_section = ".ivt.interrupts")]
173        static __INTERRUPTS: [$crate::InterruptVector; $crate::INTERRUPT_COUNT] = {
174            const fn make_array<const N: usize, const M: usize>(
175                prefix: [$crate::InterruptVector; M],
176            ) -> [$crate::InterruptVector; N] {
177                assert!(M <= N, "Prefix length cannot exceed array length");
178                let mut arr = [$crate::InterruptVector::DEFAULT; N];
179                let mut i: usize = 0;
180                while i < M {
181                    arr[i] = prefix[i];
182                    i += 1;
183                }
184                arr
185            }
186
187            make_array([$($elems),*])
188        };
189    };
190}
